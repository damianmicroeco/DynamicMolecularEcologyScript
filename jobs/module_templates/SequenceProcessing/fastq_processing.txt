#
# Sequence Processing
#

APPTAINER_SEQPROCESSING="apptainer exec {{APPTAINER_DIR}}/SequenceProcessing.sif"

#Create directory for fastqc output
mkdir {{FASTQ_OUTPUT_DIR}}/fastq_out_fastqc

#Do the fastqc
$APPTAINER_SEQPROCESSING fastqc -o {{FASTQ_OUTPUT_DIR}}/fastq_out_fastqc {{SEQUENCING_DATA}}/*.gz

#Run multiqc
mkdir {{FASTQ_OUTPUT_DIR}}/fastq_out_multiqc_raw
$APPTAINER_SEQPROCESSING multiqc {{FASTQ_OUTPUT_DIR}}/fastq_out_fastqc --outdir {{FASTQ_OUTPUT_DIR}}/fastq_out_multiqc_raw

#Make directory for storing TrimGalore output
mkdir {{FASTQ_OUTPUT_DIR}}/fastq_out_trimgal

#create a fastq file list for trimming
for f in {{SEQUENCING_DATA}}/*${READ1SUFFIX}.fastq.gz; do
    printf '%s\n' "${f%${READ1SUFFIX}.fastq.gz}"
done > {{FASTQ_OUTPUT_DIR}}/fastq_out_fastq_list.txt

#Run trim_galore variable and do a while loop for paired-end files
#20 is used for --2colour because it is the default phred score for the -q, which is the standard quality cutoff
while read i
do
    echo "start trimming $i"
    $APPTAINER_SEQPROCESSING trim_galore --fastqc --2colour {{PHRED}} --cores {{CORES}} -o {{FASTQ_OUTPUT_DIR}}/fastq_out_trimgal --paired ${i}${READ1SUFFIX}.fastq.gz ${i}${READ2SUFFIX}.fastq.gz
    echo "done trimming $i"
done<{{FASTQ_OUTPUT_DIR}}/fastq_out_fastq_list.txt

#Make output directory (done)
mkdir {{FASTQ_OUTPUT_DIR}}/fastq_out_multiqc_trimmed

#Run multiqc
$APPTAINER_SEQPROCESSING multiqc {{FASTQ_OUTPUT_DIR}}/fastq_out_trimgal --outdir {{FASTQ_OUTPUT_DIR}}/fastq_out_multiqc_trimmed

#create a mapping file to map the universal naming structure of this program with the original files
count=1
>{{FASTQ_OUTPUT_DIR}}/sample_name_map.tsv
for file1 in {{FASTQ_OUTPUT_DIR}}/fastq_out_trimgal/*${READ1SUFFIX}_val_1.fq.gz; do
    #Check if files matching the pattern actually exist
    [ -e "$file1" ] || continue
    
    #Extract the base prefix by removing the Read 1 suffix
    prefix="${file1%${READ1SUFFIX}_val_1.fq.gz}"
    
    #Define the expected matching paired file
    file2="${prefix}${READ2SUFFIX}_val_2.fq.gz"
    
    #Only proceed if the matching paired file exists
    if [ -f "$file2" ] || [ -e "$file2" ]; then
        # Format the count to be zero-padded and 4 digits
        printf -v padded_count "%04d" "$count"

        # Define the new sample name replacement
        sample_name="{{FASTQ_OUTPUT_DIR}}/fastq_out_trimgal/sample${padded_count}"
        
        # Generate the new filenames
        new_file1="${sample_name}_1_val_1.fq.gz"
        new_file2="${sample_name}_2_val_2.fq.gz"
        
        # Append the tab-delimited mapping to the output file
        printf "%s\t%s\t%s\t%s\n" "$file1" "$file2" "$new_file1" "$new_file2" >> {{FASTQ_OUTPUT_DIR}}/sample_name_map.tsv
        
        # Rename the physical files
        mv "$file1" "$new_file1"
        mv "$file2" "$new_file2"
        
        # Increment the counter for the next pair
        ((count++))
    fi
done

echo "!!!!!!!!!!!!!!!!!!!!!!!!!!!!!"
