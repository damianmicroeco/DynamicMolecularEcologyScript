APPTAINER_MOSDEPTH="apptainer exec {{APPTAINER_DIR}}/Coverage.sif"

#get coverage for all samples. Default is windows of 100 bp and quality thresholds of 30. Please feel free to change to values that fit your question
mkdir {{PROJECTDIR}}/mosdepth_out/
for i in $(ls {{PROJECTDIR}}/bwa_align/*_analysis_ready.bam)
do
    #echo $i
    j="${i##*/}"
    #echo $j
    $APPTAINER_MOSDEPTH mosdepth -t {{CORES}} {{PROJECTDIR}}/mosdepth_out/$j -b {{WINDOWSIZE}} -Q {{QUALITY}} $i
done

#gunzip all of the regions.bed files
find . -regextype posix-extended -regex '{{PROJECTDIR}}/mosdepth_out/*regions.bed\.gz' -exec gunzip {} +


#Visualize coverage across whole genome using coverage.R script
$APPTAINER_MOSDEPTH Rscript {{TEMPLATE_DIR}}/Coverage/coverage.R \
    -i {{PROJECTDIR}}/mosdepth_out \
    -g {{GTF}} \
    -o {{OUTPUT_DIR}}

    #-i: Directory where mosdepth outputs are stored
    #-g: GTF file of genome 
    #-o: Directory for storing outputs of mosdepth coverage visualization
